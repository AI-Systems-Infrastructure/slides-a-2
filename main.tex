\begin{frame}{Talking to APIs in a Python program}
  \textbf{Focus of this module:} The \texttt{requests} package in Python

  \begin{block}{What is \texttt{requests}?}
    The de facto Python package for talking with HTTP APIs.
  \end{block}

  \begin{block}{Why not alternatives?}
    AI providers ship their own Python packages such as \texttt{openai} and \texttt{anthropic}.
  \end{block}
\end{frame}


\begin{frame}[fragile]{Why not AI provider packages?}
    Good for shipping production code, bad for learning.
    \begin{itemize}
      \item SDKs \textit{hide the HTTP details} we want to get familiar with
      \item HTTP details are generalizable knowledge that will come in handy when we \textbf{build our own AI systems}
    \end{itemize}

    \begin{minipage}[t]{0.45\textwidth}
        With \texttt{openai}
        \begin{minted}[fontsize=\scriptsize]{python}
from openai import OpenAI
client = OpenAI()
response = client.chat.completions.create(
    model="gpt-5.4-mini",
    messages=[{"role": "user",
               "content": "Hello!"}],
)
        \end{minted}
    \end{minipage}
    \hfill
    \begin{minipage}[t]{0.52\textwidth}
        With \texttt{requests}
        \begin{minted}[fontsize=\scriptsize]{python}
import requests
response = requests.post(
    "https://api.openai.com/v1/chat/completions",
    headers={"Authorization": f"Bearer {key}",
             "Content-Type": "application/json"},
    json={"model": "gpt-5.4-mini",
          "messages": [{"role": "user",
                        "content": "Hello!"}]},
)
        \end{minted}
    \end{minipage}

\section{Sending HTTP Requests}

\begin{frame}{Sending HTTP Requests with \texttt{requests}}
  Recall the three parts of an \textbf{HTTP request:} request line (method, URL, and protocol version), headers, and body.

    \begin{itemize}
        \item \texttt{requests} has one function per HTTP method, like \texttt{requests.get} and \texttt{requests.post}
        \item URL passes as the first argument
        \item Headers and body pass as keyword arguments
        \item Protocol version is chosen for us
    \end{itemize}
\note{
}
\end{frame}


\begin{frame}[fragile]{Sending HTTP Requests with \texttt{requests}}{Example \texttt{GET} request}
    \begin{minted}{python}
import requests

url = "https://api.openai.com/v1/models"
headers = {
    "Authorization": "Bearer sk-abc1234567890qwerty",
    "Accept": "application/json",
    "User-Agent": "SomeAIApp/1.0",
}

response = requests.get(url, headers=headers)
    \end{minted}
\note{
}
\end{frame}


\begin{frame}[fragile]{Sending HTTP Requests with \texttt{requests}}{Example \texttt{POST} request}
    \begin{minted}{python}
url = "https://api.openai.com/v1/chat/completions"
headers = {
    "Authorization": "Bearer sk-abc1234567890qwerty",
    "Content-Type": "application/json",
}
body = {
    "model": "gpt-5.4-mini",
    "messages": [{"role": "user",
                  "content": "Write a haiku about APIs."}],
    "temperature": 0.7,
}

response = requests.post(url, headers=headers, json=body)
    \end{minted}
\note{
}
\end{frame}



\section{Handling HTTP Responses}

\begin{frame}[fragile]{The response object}
  The \texttt{response} object exposes all three parts of the \textbf{HTTP response} (i.e., status line, headers, and body).

    \begin{minted}{python}
response.status_code   # 200, 401, 500, ...
response.reason        # "OK", "Not Found", ...
response.ok            # True if status < 400

response.headers["Content-Type"]   # like a Python dict

response.text     # body as a string
response.content  # body as raw bytes
response.json()   # body parsed from JSON to a Python dict
    \end{minted}
\note{
}
\end{frame}


\begin{frame}[fragile]{Reading the response body}
    Once we parse the response body into a Python dict, the rest is normal Python.
    \begin{minted}{python}
result = response.json()
print(result["choices"][0]["message"]["content"])
    \end{minted}

    \vspace{0.5em}
    \begin{itemize}
        \item The field names depend on the API
        \item Always check the provider's documentation for the exact shape
    \end{itemize}
\note{
}
\end{frame}


\begin{frame}{Error handling}
    \texttt{requests} signals failures in two ways.

    \begin{block}{Network-level failures}
        \begin{itemize}
          \item The request failed to reach the API provider (e.g., connection refused, DNS error, timeout)
          \item \texttt{requests} will raise exceptions from \texttt{requests.exceptions}
        \end{itemize}
    \end{block}

    \begin{block}{HTTP-level failures}
        \begin{itemize}
          \item The request reaches the server, but the server responds with a \texttt{4xx} or \texttt{5xx} status code
          \item Check \texttt{response.status\_code} ourselves, or \dots
          \item No exception by default; we can call \texttt{response.raise\_for\_status()}
        \end{itemize}
    \end{block}
\note{
}
\end{frame}


\begin{frame}[fragile]{Error handling}{Example full block}
    \begin{minted}{python}
try:
    response = requests.post(url, headers=headers, json=body, timeout=30)
    response.raise_for_status()
    result = response.json()
    print(result["choices"][0]["message"]["content"])
except requests.exceptions.Timeout:
    print("The API took too long to respond.")
except requests.exceptions.HTTPError as e:
    print(f"The API returned an error: {e}")
    print(f"Body: {response.text}")
except requests.exceptions.RequestException as e:
    print(f"Request failed: {e}")
    \end{minted}
\note{
}
\end{frame}



\section{Beyond the Basics}

\begin{frame}{Three common patterns}
Good practices that we should consider when integrating AI API interaction into a real Python program.

    \begin{itemize}
\item Keeping \textbf{API keys} out of source code
\item Sending \textbf{images} to AI APIs that accept them
\item Receiving \textbf{streamed responses}
    \end{itemize}
\note{
}
\end{frame}


\begin{frame}[fragile]{Storing API keys in environment variables}
    \begin{block}{Why not hardcode API keys in code?}
\begin{enumerate}
\item API keys give whoever holds them access to your account and billing, so they should be \textbf{treated like passwords}
\item \textit{Hardcoded keys leak}, since people may commit them to public repos, paste them in WhatsApp, or screenshot them in tutorials
\item Bots scan public commits, and stolen keys can be \textit{running up bills} on your account within minutes
\end{enumerate}
    \end{block}
\note{
}
\end{frame}


\begin{frame}[fragile]{Storing API keys in environment variables}
  \begin{block}{Write environment variables (in Linux and macOS)}
        \begin{minted}{bash}
export OPENAI_API_KEY="sk-abc1234567890qwerty"
        \end{minted}
    \end{block}

    \begin{block}{Read environment variables}
        \begin{minted}{python}
import os
headers = {
    "Authorization": f"Bearer {os.getenv('OPENAI_API_KEY')}",
}
        \end{minted}
    \end{block}
\note{
}
\end{frame}


\begin{frame}[fragile]{Storing API keys in environment variables}
  For projects with many/long-term secrets: a \texttt{.env} file plus \texttt{python-dotenv}.
  \begin{block}{Write a \texttt{.env} file}
        \begin{minted}{text}
OPENAI_API_KEY=sk-abc1234567890qwerty
        \end{minted}
    \end{block}

    \begin{block}{Read environment variables}
        \begin{minted}{python}
import os
from dotenv import load_dotenv  # Third-party package

load_dotenv()  # Python won't read the `.env` file by default
headers = {
    "Authorization": f"Bearer {os.getenv('OPENAI_API_KEY')}",
}
        \end{minted}
    \end{block}

\note{
}
\end{frame}


\begin{frame}[fragile]{Sending images in the request body}
    \begin{block}{What's special about fitting images in the body?}
\begin{enumerate}
\item Many modern AI chatbot APIs accept images in addition to text
\item \textit{The catch} is that JSON cannot directly hold raw image bytes
\item \textbf{The standard workaround} is to encode the image into a \texttt{string}
\end{enumerate}
    \end{block}
\note{
}
\end{frame}


\begin{frame}[fragile]{Sending images in the request body}{Encode images as strings}
  Turns the raw bytes of images into a \texttt{string} that fits inside a JSON body.
    \begin{minted}{python}
import base64
from io import BytesIO
from PIL import Image

image = Image.open("cat.jpg")

buffer = BytesIO()
image.save(buffer, format="JPEG")
encoded = base64.b64encode(buffer.getvalue()).decode("utf-8")
    \end{minted}
\note{
}
\end{frame}


\begin{frame}[fragile]{Sending images in the request body}{Putting the image in the body}
    \begin{minted}{python}
body = {
    "model": "gpt-5.4-mini",
    "messages": [{
        "role": "user",
        "content": [
            {"type": "text", "text": "What is in this picture?"},
            {"type": "image_url",
             "image_url": {"url": f"data:image/jpeg;base64,{encoded}"}},
        ],
    }],
}

response = requests.post(url, headers=headers, json=body)
    \end{minted}
\note{
}
\end{frame}


\begin{frame}{Streaming responses with SSE}
\begin{minipage}[t]{0.47\textwidth}
\begin{enumerate}
\item Most AI apps stream responses, i.e., the response appears word by word as the AI model generates it
\item Streaming makes responses feel more responsive; otherwise, a 30-second answer takes 30 seconds before the user sees anything
\item This is achievable through the same APIs, using \emphasis{Server-Sent Events (SSE)} on top of HTTP
\end{enumerate}
\end{minipage}
\hfill
\begin{minipage}[t]{0.5\textwidth}
    \begin{figure}[htpb]
        \begin{center}
            \includegraphics[width=0.9\textwidth]{streaming-comparison}
        \end{center}
    \end{figure}
\end{minipage}

\note{
}
\end{frame}


\begin{frame}{Server-Sent Events (SSE)}
SSE is built on top of HTTP.

\begin{enumerate}
  \item \textit{Without SSE}, the server sends one complete response and closes the connection
  \item \textbf{With SSE}, the server keeps the connection open and continues to deliver the response in small chunks as they become ready
\end{enumerate}

    \begin{figure}[htpb]
        \begin{center}
            \includegraphics[width=0.6\textwidth]{sse}
        \end{center}
    \end{figure}
\note{
}
\end{frame}


\begin{frame}{Enabling SSE with \texttt{requests}}
\begin{enumerate}
\item \textbf{Tell the API we want streaming:} The exact mechanism varies by API provider
\item \textbf{Tell \texttt{requests} not to buffer:} Pass \texttt{stream=True} to the function so we can read chunks as they arrive
\end{enumerate}

\note{
}
\end{frame}


\begin{frame}[fragile]{Reading the response chunk by chunk}
    \begin{minted}{python}
body["stream"] = True
response = requests.post(url, headers=headers, json=body,
                         stream=True, timeout=60)

for line in response.iter_lines():  # iterate incoming chunks
    if not line: continue  # empty line
    line = line.decode("utf-8")
    if not line.startswith("data: "): continue  # practically empty line
    payload = line[len("data: "):]
    if payload == "[DONE]": break  # Stream ends
    # Extract actual content
    chunk = json.loads(payload)
    delta = chunk["choices"][0]["delta"].get("content", "")
    print(delta, end="", flush=True)
    \end{minted}
\note{
}
\end{frame}

\begin{frame}{A command line chatbot}
    Build a Python program that takes a question from you, sends it to an AI API, and prints the model's reply.

    Knowledge and resources from Module A.1 exercise can be reused.

    \begin{block}{Outline}
        \begin{enumerate}
            \item Read the API key from an environment variable
            \item Take a question from the command line

\end{frame}


\begin{frame}{Extensions}
    Once the basic version works, try the following to see what happens.

    \begin{enumerate}
